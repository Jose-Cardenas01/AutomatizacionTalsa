CLIENTES (
    NIT INT IDENTITY(1,1) PRIMARY KEY,
    CODIGO AS ('CLI-' + RIGHT('00' + CAST(ID AS NVARCHAR(10)),
                CASE
                    WHEN ID < 100 THEN 3
                    ELSE LEN(CAST(ID AS NVARCHAR(10)))
                END
                )) PERSISTED UNIQUE,
    NOMBRE NVARCHAR(100) NOT NULL,
    INDUSTRIA NVARCHAR(50),
    ID_UBICACION INT,
    DESCRIPCION NVARCHAR(200)
    CONTRAINT fk_clientes_ubicacion FK (ID_UBICACION) REFERENCIA UBICACIONES(ID)
)

VENTAS (
    id_venta INT IDENTITY(1,1) PRIMARY KEY,
    codigo_cliente NVARCHAR(20),
    fecha_venta DATE,
    monto_usd DECIMAL(18,2),
    id_ubicacion int
    CONTRAINT fk_ventas_clientes FK (codigo_cliente) REFERENCE CLIENTES(CODIGO)
    CONTRAINT fk_ventas_ubicacion FK (id_ubicacion) REFERENCIA UBICACIONES(ID)
)

Detalles_Ventas (
    ID INT IDENTITY(1,1) PRIMARY KEY,
    ID_VENTA INT,
    ID_EQUIPO INT,
    CANTIDAD INT,
    PRECIO_UNITARIO DECIMAL(18,2)
    CONTRAINT fk_dt_ventas_VENTAS FK (ID_VENTA) REFERENCE VENTAS(id_venta)
    CONTRAINT fk_dt_ventas_EQUIPOS FK (ID_EQUIPO) REFERENCE EQUIPOS(ID)
)

EQUIPOS (
    ID INT IDENTITY(1,1) PRIMARY KEY,
    MODELO NVARCHAR(20),
    NOMBRE NVARCHAR(60),
    PRECIO DECIMAL(18,2)
)
